# Pass 4: protocol version 2 (app/hindsight.h): BOX, BAR, TEXT's background,
# the sixteen colours, LED, BATCH, and version 1 beside it. Every expect,
# pixel, band and light must hold, or the run fails.
boot
wait 1500
sysex F0 7D 48 53 01 01 00 F7              # HELLO in version 1: version 2, 96 items
expect F0 7D 48 53 01 01 7F 00 02 60 F7
sysex F0 7D 48 53 02 02 00 F7              # and in version 2, answered in version 2
expect F0 7D 48 53 02 02 7F 00 02 60 F7
# Version 1 as it always was: no BOX, seven colours, TEXT on CLEAR's colour.
clear 1
expect F0 7D 48 53 01 01 7F 00 F7
sysex F0 7D 48 53 01 03 06 00 00 00 00 0A 00 0A 00 02 F7
expect F0 7D 48 53 01 03 7F 01 F7          # BOX: unknown to version 1
sysex F0 7D 48 53 01 04 01 07 F7
expect F0 7D 48 53 01 04 7F 02 F7          # colour 7 is version 2's
text 8 8 1 2 V1
wait 200

pixel 13 8 18E3                            # and the gap after it CLEAR's colour
